# M4 GUI Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 完成 M4：以 PySide6 打造完整 GUI——主視窗（腳本/規則清單、執行控制、即時日誌）、腳本步驟編輯器、規則編輯器、錄製面板、區域框選器（兼模板擷取）、設定頁（熱鍵/OCR 語言/日誌等級/macOS 權限狀態）；`Player`/`Recorder`/`RuleEngine` 於 QThread 執行，GUI 全程可操作（spec §7/§9）。同時吸收 M3 最終審查「M4 計畫必須吸收的待辦」全部：Minor #1 驗證訊息 label 化、M6 GUI-readiness 缺口（`on_step`、`on_fire`、storage delete/list、wait 輪詢例外政策）、連續失敗 N 次通知、`find_windows` getAppName 效能、TemplateStore missing/corrupt 區分與存檔命名重試（GUI 對話框自然吸收）。

**Architecture:** 每個 GUI 畫面一律切成「純 Python controller/model（無 Qt import，可完整單元測試）＋薄 QWidget（只做版面與 signal/slot 接線）」兩層；Qt import 只允許出現在 `app/`，core/platforms 完全不知道 Qt。GUI 與 core 之間唯一的執行緒橋樑是 `app/worker.py` 的 `Worker`（job 搬上 QThread、callback 轉 signal），core 只透過 M4 前置任務加上的注入 callback（`Player.on_step`、`RuleEngine.on_fire/on_error`）回報進度。Qt 測試以 `QT_QPA_PLATFORM=offscreen` 無頭執行（`tests/app/conftest.py` 提供 session 級 QApplication），不引入 pytest-qt。

**Tech Stack:** Python 3.11+、PySide6（>=6.8，撰稿時 dry-run 實測解為 6.11.1，cp310-abi3 wheel 相容本 venv 的 Python 3.14.6）、pynput / pywinctl / mss / opencv-python-headless / numpy（沿用）、pyobjc（Quartz 權限 preflight）、pytest。

**對應 spec:** `docs/superpowers/specs/2026-07-06-keyboard-executor-design.md` §9（六個畫面：主視窗→Task 4、錄製面板→Task 8、腳本編輯器→Task 5、規則編輯器→Task 6、區域框選器→Task 7、設定頁→Task 8）、§7（QThread 回放、緊急停止雙保險→Task 3/4）、§10（錯誤處理：OCR/截圖失敗跳過該輪＋連續失敗 N 次通知→Task 1/8；macOS 權限引導→Task 2；JSON 損壞報錯不崩潰→各編輯器）、§13-M4。吸收 `docs/superpowers/notes/2026-07-06-m3-final-review.md`「M4 計畫必須吸收的待辦」1、2(a)–(d)、4、5、6 全部（見各任務）；待辦 3（I4：watch 觸發腳本忽略 game_input_mode）屬 Windows live backlog，維持 M3 審查裁定**延至 M5-prep**，本計畫不動。

- Qt 測試一律無頭：`tests/app/conftest.py` 於模組層設 `QT_QPA_PLATFORM=offscreen`（在第一次 import PySide6 widgets 前生效），並提供 session 級 `qapp` fixture（`QApplication.instance() or QApplication([])`）。conftest 放在 `tests/app/` 下，fixture 與環境變數不影響既有非 GUI 測試
- **不引入 pytest-qt**。直接驅動 widget：呼叫 slot 方法、設值、發 signal、檢查 model 狀態；跨執行緒 signal 以 `qapp.processEvents()` 迴圈等待

```python
"""Player on_step 進度回呼（M3 審查 M6(a)：GUI 日誌面板的掛點）。"""
from keyboard_executor.core.models import script_from_dict
from keyboard_executor.core.player import Player
from keyboard_executor.platforms.base import TextBox, WindowInfo
from keyboard_executor.platforms.factory import Platform
from tests.fakes import FakeInputBackend, FakeOcrEngine, FakeScreenCapture, FakeWindowManager


def make_window():
    return WindowInfo(handle=object(), title="記事本", app_name="App",
                      x=0, y=0, width=200, height=100, is_active=False)


def make_platform(capture=None, ocr=None):
    backend = FakeInputBackend()
    manager = FakeWindowManager([make_window()])
    return Platform(input_backend=backend, window_manager=manager,
                    screen_capture=capture, ocr_engine=ocr), backend


def make_script(actions):
    return script_from_dict({"name": "測試", "target_window": {"title_pattern": "記事本"},
                             "actions": actions})


def test_on_step_receives_ordered_records():
    records = []
    platform, _ = make_platform()
    script = make_script([
        {"type": "key_press", "key": "a"},
        {"type": "delay", "ms": 0},
        {"type": "key_press", "key": "b"},
    ])
    result = Player(platform, sleep=lambda s: None, on_step=records.append).play(script)
    assert result.ok
    assert [(r["index"], r["total"], r["type"]) for r in records] == [
        (0, 3, "key_press"), (1, 3, "delay"), (2, 3, "key_press")]
    assert all(r["kind"] == "step" for r in records)
    assert records[0]["label"] == "按鍵 a"
    assert records[1]["label"] == "延遲 0ms"


def test_on_step_branch_local_index_and_total():
    records = []
    platform, _ = make_platform()
    script = make_script([
        {"type": "loop", "count": 2, "actions": [
            {"type": "key_press", "key": "x"},
            {"type": "key_press", "key": "y"},
        ]},
    ])
    result = Player(platform, sleep=lambda s: None, on_step=records.append).play(script)
    assert result.ok
    # 外層 loop 一筆＋每輪兩筆 ×2；index/total 為該層動作清單的（GUI 顯示 [n/N]）
    assert [(r["index"], r["total"], r["type"]) for r in records] == [
        (0, 1, "loop"),
        (0, 2, "key_press"), (1, 2, "key_press"),
        (0, 2, "key_press"), (1, 2, "key_press")]


def test_on_step_reported_before_wait_polling():
    records = []
    cap = FakeScreenCapture(scale=1.0)
    ocr = FakeOcrEngine(boxes=[TextBox(text="完成", x=0, y=0, width=10, height=10,
                                       confidence=0.9)])
    platform, _ = make_platform(capture=cap, ocr=ocr)
    script = make_script([{"type": "wait_text", "pattern": "完成"}])
    result = Player(platform, sleep=lambda s: None, on_step=records.append).play(script)
    assert result.ok
    assert [r["type"] for r in records] == ["wait_text"]
    assert records[0]["label"] == "等待文字「完成」"


def test_on_step_exception_aborts_as_unexpected_error():
    def boom(record):
        raise RuntimeError("GUI 回呼壞掉")

    platform, backend = make_platform()
    script = make_script([{"type": "key_press", "key": "a"}])
    result = Player(platform, sleep=lambda s: None, on_step=boom).play(script)
    assert result.ok is False
    assert "未預期錯誤" in result.error
    assert backend.calls == []   # 回呼於動作執行前，動作未送出
```

```python
             default: int | None = None, required: bool = True) -> int:
    if name not in params:
        if required:
            _err(f"{label} 缺少欄位 {name}")
        return default
    value = params[name]
    if not isinstance(value, int) or isinstance(value, bool):

    if minimum is not None and value < minimum:
```

```python
def _get_region(params: dict, label: str):
    """region：None（整個視窗）或 [x, y, w, h] 四整數（視窗相對，w/h > 0）。"""
    region = params.get("region")
    if region is None:
        return None
    ok = (isinstance(region, (list, tuple)) and len(region) == 4
          and all(isinstance(v, int) and not isinstance(v, bool) for v in region)
          and region[2] > 0 and region[3] > 0)
    if not ok:
        _err(f"{label} 欄位 region 必須是 null 或 [x, y, w, h] 四整數（w、h > 0）")
    return [region[0], region[1], region[2], region[3]]
```

```python
    def _poll_until(self, matcher, timeout_ms) -> tuple[int, int] | None:
        """輪詢 matcher() 直到回傳座標或逾時；stop_event 於 _sleep 途中提早醒來。
        逾時回傳 None。matcher 的暫時性例外（擷取/OCR/比對失敗）視為本輪不中：
        記錄並跳過該輪（spec §10），不中止整個回放；_PlayerAbort（缺能力、
        視窗關閉）仍上拋中止。"""
        deadline = self._clock() + timeout_ms / 1000
        while True:
            self._check_stop()
            try:
                hit = matcher()
            except _PlayerAbort:
                raise
            except Exception as e:
                print(f"wait 動作本輪擷取或辨識失敗，已跳過：{e}", file=sys.stderr)
                hit = None
            if hit is not None:
                return hit
            if self._clock() >= deadline:
                return None
            self._sleep(_POLL_INTERVAL_MS / 1000)
```

```python
from __future__ import annotations

import sys


def _default_permission_checks():
    """darwin 三權限檢查（輔助使用/螢幕錄製/輸入監控）；非 darwin 回傳空 dict。
    sys.platform 判斷沿用 cli._check_darwin_permissions 的已核准例外模式。"""
    if sys.platform != "darwin":
        return {}
    from keyboard_executor.platforms.darwin.permissions import (
        ACCESSIBILITY_GUIDE,
        INPUT_MONITORING_GUIDE,
        SCREEN_RECORDING_GUIDE,
        check_accessibility,
        check_input_monitoring,
        check_screen_recording,
    )

    return {
        "輔助使用": (check_accessibility, ACCESSIBILITY_GUIDE),
        "螢幕錄製": (check_screen_recording, SCREEN_RECORDING_GUIDE),
        "輸入監控": (check_input_monitoring, INPUT_MONITORING_GUIDE),
    }


def collect_permission_warnings(checks=None) -> list[str]:
    """回傳缺失權限的繁中引導文字清單（純邏輯，checks 注入供測試）。
    個別檢查拋例外時視同缺失——保守提示使用者，不讓啟動崩潰（spec §10）。"""
    checks = checks if checks is not None else _default_permission_checks()
    warnings = []
    for _name, (check, guide) in checks.items():
        try:
            ok = check()
        except Exception:
            ok = False
        if not ok:
            warnings.append(guide)
    return warnings


def build_app(argv=None, *, permission_checks=None):
    """建立（或重用）QApplication 與主視窗。Qt import 延後到函式內，
    讓純邏輯（collect_permission_warnings）可在無 Qt 環境單獨測試。"""
    from PySide6.QtWidgets import QApplication

    from keyboard_executor.app.main_window import MainWindow

    app = QApplication.instance() or QApplication(list(argv) if argv else [])
    window = MainWindow(
        permission_warnings=collect_permission_warnings(permission_checks))
    return app, window
```

### Task 3: Worker——QThread 執行緒橋（可測）

GUI 保持可操作的脊椎（spec §7）：`Worker` 把一個「job」（接收 progress callback 的 callable）搬上 QThread，並以 Qt signal 把進度/結果/失敗送回 GUI 執行緒。core 完全不知道 Qt——`Player.on_step`/`RuleEngine.on_fire`（Task 1）只是普通 callback，由 job 內部接到 Worker 給的 `progress` 函式上（接線在 Task 4/8 的 controller）。signal 契約在此鎖定：`progress(dict)`（payload 即跨任務契約的 kind 記錄）、`finished(object)`（job 回傳值原樣）、`failed(str)`（繁中訊息）。橋接邏輯以假 job 測試：`run()` 直呼＝同執行緒同步（direct connection，不需事件迴圈）；`start()` 的真實跨執行緒投遞以 `qapp.processEvents()` 迴圈驗證一次。

**切分：** 可測核心＝`Worker.run()` 的「job 例外→failed、回傳值→finished、progress 轉發順序」契約；執行緒面＝`start()` 一個真跨執行緒測試；真實 Player/RuleEngine 不進本任務測試。

**Files:**
- Create: `src/keyboard_executor/app/worker.py`
- Test: `tests/app/test_worker.py`（新檔，6 個）

**Interfaces:**
- Consumes: PySide6 `QObject`/`QThread`/`Signal`

```python
"""QThread 工作橋（spec §7：回放/監看於背景執行緒執行，GUI 保持可操作）。

Worker 把一個「job」（接收 progress callback 的 callable）搬上 QThread，
並以 Qt signal 送回進度/結果/失敗——這是 GUI 與 core 之間唯一的執行緒橋樑：
core（Player/RuleEngine）完全不知道 Qt，只透過注入 callback（on_step/on_fire/
notifier/on_error）回報；job 把這些 callback 接到 Worker 提供的 progress 上，
Worker 再以跨執行緒安全的 signal emit 投遞回 GUI 執行緒。

job 契約：job(progress: Callable[[dict], None]) -> Any。progress 收到的 dict
原樣成為 progress signal 的 payload（kind="step"/"fire"/"notify"/"error"，
見計畫跨任務契約）。Worker 為一次性：一個 job 一個 Worker，不可重複 start。
"""
from __future__ import annotations

from PySide6.QtCore import QObject, QThread, Signal


class Worker(QObject):
    progress = Signal(dict)
    finished = Signal(object)
    failed = Signal(str)

    def __init__(self, job, parent=None):
        super().__init__(parent)
        self._job = job
        self._thread: QThread | None = None

    def start(self) -> None:
        """把自己搬上新 QThread 並執行 run()；一次性，重複 start 拋錯。"""
        if self._thread is not None:
            raise RuntimeError("Worker 已啟動，不能重複啟動")
        self._thread = QThread()
        self.moveToThread(self._thread)
        self._thread.started.connect(self.run)
        self._thread.start()

    def run(self) -> None:
        """執行 job。直接呼叫＝在當前執行緒同步執行（測試/ImmediateWorker 用）；
        經 start() 呼叫＝在背景 QThread 執行，signal 以 queued connection 投遞。"""
        try:
            result = self._job(self.progress.emit)
        except Exception as e:  # job 例外不可讓背景執行緒無聲死亡
            self.failed.emit(f"執行失敗：{e}")
        else:
            self.finished.emit(result)
        finally:
            if self._thread is not None:
                self._thread.quit()

    def wait(self, timeout_ms: int = 5000) -> bool:
        """等待背景執行緒結束（測試與關閉流程用）；未曾 start 視為已結束。"""
        if self._thread is None:
            return True
        return self._thread.wait(timeout_ms)
```

```python
"""腳本清單狀態與摘要（純 Python，無 Qt import）。"""
from __future__ import annotations

from keyboard_executor.core.models import ScriptValidationError


class ScriptListController:
    def __init__(self, storage):
        self._storage = storage
        self._names: list[str] = []
        self._selected: str | None = None

    def refresh(self) -> list[str]:
        self._names = self._storage.list_names()
        if self._selected not in self._names:
            self._selected = None
        return list(self._names)

    @property
    def names(self) -> list[str]:
        return list(self._names)

    def select(self, name: str) -> None:
        if name not in self._names:
            raise ValueError(f"未知腳本: {name!r}")
        self._selected = name

    @property
    def selected(self) -> str | None:
        return self._selected

    def delete_selected(self) -> str:
        if self._selected is None:
            raise ValueError("尚未選取腳本")
        name = self._selected
        self._storage.delete(name)
        self.refresh()
        return name


def summary_for_script(storage, name: str) -> str:
    """右側摘要文字；載入失敗回繁中錯誤字串（不拋，GUI 直接顯示）。"""
    try:
        script = storage.load(name)
    except ScriptValidationError as e:
        return f"腳本載入失敗：{e}"
    target = script.target_window.title_pattern or script.target_window.app_name
    repeat = "無限" if script.settings.repeat == -1 else f"{script.settings.repeat} 次"
    return (f"腳本「{script.name}」：{len(script.actions)} 個步驟，"
            f"目標視窗 {target}，重複 {repeat}")
```

```python
    def build_job(self, name: str):
        """載入腳本並回傳 Worker job。載入失敗（不存在/損毀）就地拋
        ScriptValidationError——發生在 GUI 執行緒，呼叫端以日誌/訊息框顯示。"""
        script = self._storage.load(name)
        stop_event = threading.Event()
        self._stop_event = stop_event

        def job(progress):
            platform = self._get_platform(
                game_input_mode=script.settings.game_input_mode)
            player = self._player_factory(platform, storage=self._storage,
                                          templates=self._templates,
                                          stop_event=stop_event,
                                          on_step=progress)
            guard = self._start_guard(stop_event)
            try:
                return player.play(script)
            finally:
                if guard is not None:
                    try:
                        guard.stop()
                    except Exception:
                        pass  # 守護收尾失敗不影響回放結果
        return job

    def stop(self) -> None:
        if self._stop_event is not None:
            self._stop_event.set()

    def _start_guard(self, stop_event):
        """best-effort 緊急停止（鏡射 cli._start_play_guard）。"""
        try:
            guard = self._guard_factory(self._hook_factory(), stop_event,
                                        stop_key=self._stop_key)
            guard.start()
            return guard
        except Exception:
            print("緊急停止熱鍵無法啟用（可能缺輸入監控權限），仍可用停止按鈕",
                  file=sys.stderr)
            return None
```

```python
"""腳本步驟編輯模型（純 Python，無 Qt import）。

內部以「正規化 action dict」（action_to_dict(action_from_dict(d)) 的形狀）保存
步驟；所有寫入先過 models.action_from_dict 驗證（繁中錯誤原樣拋
ScriptValidationError 供 GUI 顯示、狀態不變），to_script() 再以 script_from_dict
做整份權威驗證——與存檔格式 round-trip 一致。"""
from __future__ import annotations

import copy

from keyboard_executor.core.models import (
    Script,
    action_from_dict,
    action_label,
    action_to_dict,
    script_from_dict,
    script_to_dict,
)


class ScriptEditorModel:
    def __init__(self, script: Script):
        data = script_to_dict(script)
        self._name: str = data["name"]
        self._target_window: dict = data["target_window"]
        self._settings: dict = data["settings"]
        self._actions: list[dict] = data["actions"]

    @property
    def name(self) -> str:
        return self._name

    def set_name(self, name: str) -> None:
        self._name = name

    def __len__(self) -> int:
        return len(self._actions)

    def rows(self) -> list[dict]:
        """[{'type': ..., 'label': ...}] 供表格顯示。"""
        return [{"type": d["type"], "label": action_label(action_from_dict(d))}
                for d in self._actions]

    def action_at(self, index: int) -> dict:
        return copy.deepcopy(self._actions[index])

    def insert(self, index: int, action_dict: dict) -> None:
        normalized = action_to_dict(action_from_dict(action_dict))
        self._actions.insert(max(0, min(index, len(self._actions))), normalized)

    def update(self, index: int, action_dict: dict) -> None:
        normalized = action_to_dict(action_from_dict(action_dict))
        self._actions[index] = normalized

    def remove(self, index: int) -> None:
        del self._actions[index]

    def duplicate(self, index: int) -> None:
        self._actions.insert(index + 1, copy.deepcopy(self._actions[index]))

    def move(self, index: int, delta: int) -> int:
        """步驟上移（delta<0）/下移（delta>0），回傳新位置（邊界夾住）。"""
        target = max(0, min(index + delta, len(self._actions) - 1))
        action = self._actions.pop(index)
        self._actions.insert(target, action)
        return target

    def to_script(self) -> Script:
        return script_from_dict({
            "name": self._name,
            "target_window": copy.deepcopy(self._target_window),
            "settings": copy.deepcopy(self._settings),
            "actions": copy.deepcopy(self._actions),
```

```python
"""Action 編輯表單的欄位規格與字串值解析（純 Python，無 Qt import）。

每個 ActionType 對應一組 (欄位名, 型別代號) 規格；表單（script_editor.ActionForm）
依此宣告式產生輸入元件，commit 時把字串值經 to_action_dict 轉為 action dict，
最終仍由 models.action_from_dict 做權威驗證。巢狀動作（loop/if 的
actions/then/else）以 JSON 文字欄位編輯（M4 取捨：視覺化巢狀編輯列 v2）。

型別代號：int、float、str、text（原樣字串，可空）、keys（"ctrl+shift+s"）、
point（"x, y"）、region（"x, y, w, h"，留空＝整個視窗）、ms（int 或 "min, max"）、
actions_json（JSON 動作清單）、choice:a/b/c（下拉）。
"""
from __future__ import annotations

import json

from keyboard_executor.core.models import ScriptValidationError

FIELD_SPECS: dict[str, list[tuple[str, str]]] = {
    "key_press": [("key", "str"), ("hold_ms", "int")],
    "key_combo": [("keys", "keys")],
    "text_input": [("text", "text")],
    "mouse_move": [("x", "int"), ("y", "int")],
    "mouse_click": [("x", "int"), ("y", "int"),
                    ("button", "choice:left/right/middle"), ("count", "int")],
    "mouse_drag": [("from", "point"), ("to", "point"),
                   ("button", "choice:left/right/middle")],
    "scroll": [("x", "int"), ("y", "int"), ("dx", "int"), ("dy", "int")],
    "delay": [("ms", "ms")],
    "loop": [("count", "int"), ("actions", "actions_json")],
    "run_script": [("script_name", "str")],
    "wait_text": [("region", "region"), ("pattern", "str"),
                  ("mode", "choice:contains/regex"), ("timeout_ms", "int"),
                  ("on_timeout", "choice:abort/continue/goto"),
                  ("goto_index", "int")],
    "wait_image": [("region", "region"), ("template", "str"),
                   ("threshold", "float"), ("timeout_ms", "int"),
                   ("on_timeout", "choice:abort/continue/goto"),
                   ("goto_index", "int")],
    "if_text": [("region", "region"), ("pattern", "str"),
                ("mode", "choice:contains/regex"),
                ("then", "actions_json"), ("else", "actions_json")],
    "if_image": [("region", "region"), ("template", "str"),
                 ("threshold", "float"),
                 ("then", "actions_json"), ("else", "actions_json")],
    "click_text": [("region", "region"), ("pattern", "str"),
                   ("mode", "choice:contains/regex")],
    "click_image": [("region", "region"), ("template", "str"),
                    ("threshold", "float")],
}


def _parse_int(name: str, raw: str) -> int:
    try:
        return int(raw)
    except ValueError:
        raise ScriptValidationError(f"欄位 {name} 必須是整數: {raw!r}") from None


def parse_int_list(name: str, raw: str, count: int) -> list[int]:
    """"1, 2" / "[1, 2]" → [1, 2]；個數不符或非整數拋 ScriptValidationError。"""
    parts = [p.strip() for p in raw.replace("[", "").replace("]", "").split(",")]
    if len(parts) != count:
        raise ScriptValidationError(
            f"欄位 {name} 必須是 {count} 個以逗號分隔的整數: {raw!r}")
    return [_parse_int(name, p) for p in parts]


def to_action_dict(atype: str, values: dict[str, str]) -> dict:
    """表單字串值 → action dict。空字串（text 除外）＝省略欄位讓 models 帶預設；
    回傳值仍需經 action_from_dict 權威驗證（呼叫端 ScriptEditorModel 會做）。"""
    if atype not in FIELD_SPECS:
        raise ScriptValidationError(f"未知動作類型: {atype!r}")
    result: dict = {"type": atype}
    for name, kind in FIELD_SPECS[atype]:
        if kind == "text":
            result[name] = values.get(name, "")
            continue
        raw = (values.get(name) or "").strip()
        if not raw:
            continue
        if kind == "int":
            result[name] = _parse_int(name, raw)
        elif kind == "float":
            try:
                result[name] = float(raw)
            except ValueError:
                raise ScriptValidationError(
                    f"欄位 {name} 必須是數字: {raw!r}") from None
        elif kind == "str" or kind.startswith("choice:"):
            result[name] = raw
        elif kind == "keys":
            result[name] = [k.strip() for k in raw.split("+") if k.strip()]
        elif kind == "point":
            result[name] = parse_int_list(name, raw, 2)
        elif kind == "region":
            result[name] = parse_int_list(name, raw, 4)
        elif kind == "ms":
            result[name] = (parse_int_list(name, raw, 2) if "," in raw
                            else _parse_int(name, raw))
        elif kind == "actions_json":
            try:
                result[name] = json.loads(raw)
            except json.JSONDecodeError as e:
                raise ScriptValidationError(
                    f"欄位 {name} 不是合法 JSON: {e}") from None
    return result
```

```python
        layout.addRow(ok)

        for i, row in enumerate(rows):
            self.table.setItem(i, 0, QTableWidgetItem(row["type"]))
            self.table.setItem(i, 1, QTableWidgetItem(row["label"]))

    def add_action(self, atype: str, values: dict[str, str]) -> None:
        """表單字串值 → 驗證 → 插入到選取列之後（無選取＝檔尾）。
        無效動作拋 ScriptValidationError（modal 處理器轉為 status_label）。"""
        action = to_action_dict(atype, values)
        index = self.table.currentRow()
        self._model.insert(len(self._model) if index < 0 else index + 1, action)
        self.refresh()

        if not ok:
            return
        form = ActionForm(atype, parent=self)
        if form.exec() != QDialog.DialogCode.Accepted:
            return
        try:
            self.add_action(atype, form.values())
            self.status_label.setText("")
        except ScriptValidationError as e:
            self.status_label.setText(f"無效的動作：{e}")

        index = self.table.currentRow()
        if index < 0:
            return
        current = self._model.action_at(index)
        atype = current.pop("type")
        form = ActionForm(atype, initial=current, parent=self)
        if form.exec() != QDialog.DialogCode.Accepted:
            return
        try:
            self._model.update(index, to_action_dict(atype, form.values()))
            self.status_label.setText("")
            self.refresh()
        except ScriptValidationError as e:
            self.status_label.setText(f"無效的動作：{e}")

    def _on_remove(self) -> None:
        index = self.table.currentRow()
        if index < 0:
            return
        self._model.remove(index)
        self.refresh()

    def _on_move(self, delta: int) -> None:
        index = self.table.currentRow()
        if index < 0:
            return
        new_index = self._model.move(index, delta)
        self.refresh()
        self.table.setCurrentCell(new_index, 0)

    def _on_duplicate(self) -> None:
        index = self.table.currentRow()
        if index < 0:
            return
        self._model.duplicate(index)
        self.refresh()

    def _on_save(self) -> None:
        self._model.set_name(self.name_edit.text())
        try:
            path = self._storage.save(self._model.to_script())
        except ScriptValidationError as e:
            self.status_label.setText(f"儲存失敗：{e}")   # 名稱不合法→留在對話框重試
            return
        self.status_label.setText(f"已儲存：{path}")
```

```python
"""規則編輯模型（純 Python，無 Qt import）。

欄位皆為「表單友善」型（字串/布林），to_rule() 組 dict 後交 rule_from_dict
權威驗證——Task 1 label 化後的「規則 …」繁中訊息可原樣顯示在 GUI。"""
from __future__ import annotations

from keyboard_executor.app.action_fields import parse_int_list
from keyboard_executor.core.models import Rule, ScriptValidationError, rule_from_dict


def _parse_number(name: str, raw: str) -> float:
    try:
        return float(raw)
    except ValueError:
        raise ScriptValidationError(f"規則欄位 {name} 必須是數字: {raw!r}") from None


class RuleEditorModel:
    def __init__(self):
        self.name = ""
        self.title_pattern = ""
        self.app_name = ""
        self.region_text = ""            # "x, y, w, h"；留空＝整個視窗
        self.condition_type = "text"     # "text" | "image"
        self.pattern = ""
        self.mode = "contains"
        self.template = ""
        self.threshold = "0.85"
        self.interval_sec = "5"
        self.cooldown_sec = "60"
        self.trigger_type = "notify"     # "notify" | "run_script" | "stop_all"
        self.script_name = ""
        self.message = ""
        self.enabled = True

    @classmethod
    def from_rule(cls, rule: Rule) -> "RuleEditorModel":
        model = cls()
        model.name = rule.name
        model.title_pattern = rule.target_window.title_pattern or ""
        model.app_name = rule.target_window.app_name or ""
        model.region_text = (", ".join(str(v) for v in rule.region)
                             if rule.region is not None else "")
        model.condition_type = rule.condition.type
        model.pattern = rule.condition.pattern or ""
        model.mode = rule.condition.mode
        model.template = rule.condition.template or ""
        model.threshold = str(rule.condition.threshold)
        model.interval_sec = str(rule.interval_sec)
        model.cooldown_sec = str(rule.cooldown_sec)
        model.trigger_type = rule.trigger.type
        model.script_name = rule.trigger.script_name or ""
        model.message = rule.trigger.message or ""
        model.enabled = rule.enabled
        return model

    def to_dict(self) -> dict:
        data: dict = {"name": self.name.strip(), "target_window": {},
                      "enabled": self.enabled}
        if self.title_pattern.strip():
            data["target_window"]["title_pattern"] = self.title_pattern.strip()
        if self.app_name.strip():
            data["target_window"]["app_name"] = self.app_name.strip()
        if self.region_text.strip():
            data["region"] = parse_int_list("region", self.region_text, 4)
        if self.condition_type == "text":
            data["condition"] = {"type": "text", "pattern": self.pattern,
                                 "mode": self.mode}
        else:
            data["condition"] = {"type": "image", "template": self.template,
                                 "threshold": _parse_number("threshold",
                                                            self.threshold)}
        data["interval_sec"] = _parse_number("interval_sec", self.interval_sec)
        data["cooldown_sec"] = _parse_number("cooldown_sec", self.cooldown_sec)
        if self.trigger_type == "run_script":
            data["trigger"] = {"type": "run_script", "script_name": self.script_name}
        elif self.trigger_type == "notify":
            data["trigger"] = {"type": "notify", "message": self.message}
        else:
            data["trigger"] = {"type": "stop_all"}
        return data

    def to_rule(self) -> Rule:
        return rule_from_dict(self.to_dict())
```

```python
"""區域框選幾何（純 Python，無 Qt import）。

拖曳座標為 Qt 邏輯座標；scale＝Qt 邏輯座標 → 平台輸入座標單位的倍率
（region_picker.drag_scale：macOS 1.0、Windows devicePixelRatio）；
win_origin 為目標視窗左上角（輸入單位，pywinctl 幾何）。"""
from __future__ import annotations

from keyboard_executor.core.geometry import Point, Rect


def normalize_drag(p1: Point, p2: Point) -> Rect:
    """任意方向拖曳的兩角 → 正規化 Rect（原點取 min、寬高取絕對值）。"""
    return Rect(min(p1.x, p2.x), min(p1.y, p2.y),
                abs(p1.x - p2.x), abs(p1.y - p2.y))


def screen_rect_to_window_region(drag_rect: Rect, win_origin: Point,
                                 scale: float = 1.0) -> list[int]:
    """螢幕拖曳矩形（Qt 邏輯座標）→ 視窗相對 region [x, y, w, h]（輸入單位）。
    寬高 <= 0 拋 ValueError（繁中）；x/y 允許負值（框選超出視窗左上）。"""
    width = round(drag_rect.width * scale)
    height = round(drag_rect.height * scale)
    if width <= 0 or height <= 0:
        raise ValueError("框選範圍太小，請重新拖曳")
    return [round(drag_rect.x * scale) - win_origin.x,
            round(drag_rect.y * scale) - win_origin.y, width, height]
```

```python
"""模板擷取編排（純 Python，無 Qt import）：框選區域 → 擷圖 → 存 PNG。"""
from __future__ import annotations

from keyboard_executor.core.geometry import Rect


def capture_template(capture, store, name: str, screen_rect_points: Rect):
    """screen_rect_points：螢幕絕對「點」矩形（drag_scale 換算後的框選結果）。
    依 capture.scale_factor() 換為實體像素（ScreenCapture 契約，見 base.py）
    後擷取，存入 TemplateStore；回傳 PNG 路徑。名稱不合法/寫入失敗由
    save_template 拋 ScriptValidationError（繁中）。"""
    scale = capture.scale_factor()
    rect = Rect(round(screen_rect_points.x * scale),
                round(screen_rect_points.y * scale),
                round(screen_rect_points.width * scale),
                round(screen_rect_points.height * scale))
    image = capture.capture(rect)
    return store.save_template(name, image)
```

```python
"""半透明全螢幕區域框選覆蓋層（spec §9）＋ Qt 邏輯座標→輸入單位倍率。

薄件：拖曳幾何純邏輯在 region_math；本檔只做覆蓋層繪製與滑鼠事件轉發
（_begin/_update/_finish 為可測 seam）。Esc 取消。region_selected 回傳的
Rect 為 Qt 邏輯座標——呼叫端以 drag_scale() 換算輸入單位後交
screen_rect_to_window_region() 得視窗相對 region。"""
from __future__ import annotations

import sys

from PySide6.QtCore import Qt, Signal
from PySide6.QtGui import QColor, QGuiApplication, QPainter, QPen
from PySide6.QtWidgets import QWidget

from keyboard_executor.app.region_math import normalize_drag
from keyboard_executor.core.geometry import Point


def drag_scale(device_pixel_ratio: float, _sys_platform=None) -> float:
    """Qt 邏輯座標 → 平台輸入座標單位的倍率（sys.platform 沿用 cli 權限檢查的
    已核准例外模式，注入供測試）：Windows 輸入單位為實體像素 →
    devicePixelRatio；macOS 為點（Qt 邏輯座標即點）→ 1.0。"""
    plat = _sys_platform if _sys_platform is not None else sys.platform
    return device_pixel_ratio if plat == "win32" else 1.0


class RegionPickerOverlay(QWidget):
    region_selected = Signal(object)   # geometry.Rect（Qt 邏輯座標，已正規化）
    cancelled = Signal()

    def __init__(self):
        super().__init__(None, Qt.WindowType.FramelessWindowHint
                         | Qt.WindowType.WindowStaysOnTopHint)
        self.setAttribute(Qt.WidgetAttribute.WA_TranslucentBackground)
        self.setCursor(Qt.CursorShape.CrossCursor)
        self._start: Point | None = None
        self._current: Point | None = None

    def open_fullscreen(self) -> None:
        screen = QGuiApplication.primaryScreen()
        self.setGeometry(screen.geometry())
        self.show()

    # -- 可測 seam（滑鼠事件轉發到這裡；座標為 Qt 邏輯的螢幕絕對座標） --

    def _begin(self, x: int, y: int) -> None:
        self._start = Point(x, y)
        self._current = Point(x, y)
        self.update()

    def _update(self, x: int, y: int) -> None:
        if self._start is None:
            return
        self._current = Point(x, y)
        self.update()

    def _finish(self, x: int, y: int) -> None:
        if self._start is None:
            return
        rect = normalize_drag(self._start, Point(x, y))
        self._start = None
        self._current = None
        self.close()
        self.region_selected.emit(rect)

    # -- Qt 事件（thin：加上覆蓋層自身位移換成螢幕座標） --

    def mousePressEvent(self, event) -> None:
        pos = event.position()
        self._begin(int(pos.x()) + self.x(), int(pos.y()) + self.y())

    def mouseMoveEvent(self, event) -> None:
        pos = event.position()
        self._update(int(pos.x()) + self.x(), int(pos.y()) + self.y())

    def mouseReleaseEvent(self, event) -> None:
        pos = event.position()
        self._finish(int(pos.x()) + self.x(), int(pos.y()) + self.y())

    def keyPressEvent(self, event) -> None:
        if event.key() == Qt.Key.Key_Escape:
            self.close()
            self.cancelled.emit()

    def paintEvent(self, event) -> None:
        painter = QPainter(self)
        painter.fillRect(self.rect(), QColor(0, 0, 0, 80))   # 半透明遮罩
        if self._start is not None and self._current is not None:
            rect = normalize_drag(self._start, self._current)
            painter.setPen(QPen(QColor(255, 60, 60), 2))
            painter.drawRect(rect.x - self.x(), rect.y - self.y(),
                             rect.width, rect.height)
```

(b) 建立 `src/keyboard_executor/app/watch_controller.py`：

```python
"""監看控制（純 Python，無 Qt import）：載入啟用規則、組 RuleEngine job。

鏡射 cli.run_watch：共用 stop_event 以 player_factory 閉包穿透給規則觸發的
Player（stop_all/F10/停止監看按鈕能中斷執行中的規則腳本）；on_fire/notifier/
on_error 全轉為 progress 記錄（kind=fire/notify/error）供 GUI 日誌；
SessionGuard best-effort（缺輸入監控權限只警告不阻斷）。"""
from __future__ import annotations

import sys
import threading

from keyboard_executor.core.guard import SessionGuard
from keyboard_executor.core.models import ScriptValidationError
from keyboard_executor.core.player import Player
from keyboard_executor.core.rule_engine import RuleEngine
from keyboard_executor.platforms.common.hook_pynput import PynputInputHook
from keyboard_executor.platforms.factory import get_platform


class WatchController:
    def __init__(self, rule_storage, script_storage, templates, *,
                 get_platform=get_platform, engine_factory=RuleEngine,
                 player_factory=Player, hook_factory=PynputInputHook,
                 guard_factory=SessionGuard, stop_key: str = "f10"):
        self._rule_storage = rule_storage
        self._script_storage = script_storage
        self._templates = templates
        self._get_platform = get_platform
        self._engine_factory = engine_factory
        self._player_factory = player_factory
        self._hook_factory = hook_factory
        self._guard_factory = guard_factory
        self._stop_key = stop_key
        self._stop_event: threading.Event | None = None

    def build_job(self):
        """載入啟用規則並回傳 Worker job；沒有啟用規則就地拋
        ScriptValidationError（GUI 執行緒，日誌顯示）。"""
        rules = self._rule_storage.load_enabled()
        if not rules:
            raise ScriptValidationError("沒有啟用中的規則")
        stop_event = threading.Event()
        self._stop_event = stop_event

        def job(progress):
            platform = self._get_platform()

            def player_factory_with_stop(*args, **kwargs):
                return self._player_factory(*args, stop_event=stop_event, **kwargs)

            engine = self._engine_factory(
                platform, storage=self._script_storage, templates=self._templates,
                notifier=lambda message: progress(
                    {"kind": "notify", "message": message}),
                on_fire=lambda name, ttype: progress(
                    {"kind": "fire", "rule": name, "trigger": ttype}),
                on_error=lambda name, message: progress(
                    {"kind": "error", "message": f"規則「{name}」：{message}"}),
                player_factory=player_factory_with_stop,
                stop_event=stop_event)
            guard = self._start_guard(stop_event)
            try:
                engine.run(rules)
            finally:
                if guard is not None:
                    try:
                        guard.stop()
                    except Exception:
                        pass  # 守護收尾失敗不影響監看結果
            return f"已停止監看（{len(rules)} 條規則）"
        return job

    def stop(self) -> None:
        if self._stop_event is not None:
            self._stop_event.set()

    def _start_guard(self, stop_event):
        """best-effort 緊急停止（鏡射 cli._start_watch_guard）。"""
        try:
            guard = self._guard_factory(self._hook_factory(), stop_event,
                                        stop_key=self._stop_key)
            guard.start()
            return guard
        except Exception:
            print("緊急停止熱鍵無法啟用（可能缺輸入監控權限），仍可用停止監看按鈕",
                  file=sys.stderr)
            return None
```

```python
"""GUI 設定（純 Python，無 Qt import）：熱鍵、OCR 語言、日誌等級。

JSON 存於 <root>/settings.json；檔案缺失→預設值；損毀→繁中警告後回預設
（spec §10 報錯不崩潰）。儲存前驗證：熱鍵必須是合法按鍵名稱（keynames）且
兩鍵不同；log_level 限 DEBUG/INFO/WARNING/ERROR。熱鍵與 OCR 語言於
控制器建構時讀取——修改後重新啟動生效（設定頁如實提示）。"""
from __future__ import annotations

import json
import sys
from dataclasses import asdict, dataclass
from pathlib import Path

from keyboard_executor.core.keynames import is_valid_key
from keyboard_executor.core.models import ScriptValidationError

_LOG_LEVELS = ("DEBUG", "INFO", "WARNING", "ERROR")


@dataclass(frozen=True)
class Settings:
    toggle_key: str = "f9"
    stop_key: str = "f10"
    ocr_language: str = "auto"
    log_level: str = "INFO"


def validate_settings(settings: Settings) -> Settings:
    for label, value in (("toggle_key", settings.toggle_key),
                         ("stop_key", settings.stop_key)):
        if not is_valid_key(value):
            raise ScriptValidationError(f"設定 {label} 不是可用的按鍵名稱: {value!r}")
    if settings.toggle_key.strip().lower() == settings.stop_key.strip().lower():
        raise ScriptValidationError("設定 toggle_key 與 stop_key 不可相同")
    if settings.log_level not in _LOG_LEVELS:
        raise ScriptValidationError(
            f"設定 log_level 必須是 {'/'.join(_LOG_LEVELS)}: {settings.log_level!r}")
    return settings


def ocr_languages_from_setting(value: str) -> list[str] | None:
    """"auto"/空 → None（引擎預設）；否則逗號分隔語言清單（如 "zh-Hant, en-US"）。"""
    value = value.strip()
    if not value or value == "auto":
        return None
    return [v.strip() for v in value.split(",") if v.strip()]


class SettingsStore:
    def __init__(self, root: str | Path):
        self._path = Path(root) / "settings.json"

    def load(self) -> Settings:
        if not self._path.exists():
            return Settings()
        try:
            data = json.loads(self._path.read_text(encoding="utf-8"))
            return validate_settings(Settings(**data))
        except Exception as e:
            print(f"設定檔載入失敗，改用預設值：{e}", file=sys.stderr)
            return Settings()

    def save(self, settings: Settings) -> Path:
        validate_settings(settings)
        try:
            self._path.parent.mkdir(parents=True, exist_ok=True)
            self._path.write_text(
                json.dumps(asdict(settings), ensure_ascii=False, indent=2) + "\n",
                encoding="utf-8")
        except OSError as e:
            raise ScriptValidationError(f"設定無法儲存: {e}") from e
        return self._path
```

```python
"""GUI 錄製會話（純 Python，無 Qt import）——把 cli.run_record 的核心編排
（dispatch 加鎖、控制鍵自我過濾、F9 切換、F10/滑鼠角落放棄）改為事件驅動
供錄製面板使用（spec §6/§9）。

狀態機：idle →（F9/toggle()）recording →（F9/toggle()）done；
stop_event（F10/角落）→ aborted，finish() 回傳 None。
on_state("recording"/"done"/"aborted") 供面板更新按鈕與狀態文字——注意
呼叫發生在 hook 的 listener 執行緒，Qt 面板必須經 signal 轉回 GUI 執行緒。"""
from __future__ import annotations

import re
import threading

from keyboard_executor.core.geometry import Point
from keyboard_executor.core.guard import SessionGuard
from keyboard_executor.core.models import (
    Script,
    ScriptSettings,
    ScriptValidationError,
    TargetWindow,
)
from keyboard_executor.core.recorder import Recorder


class RecorderSession:
    def __init__(self, hook, origin: Point, *, mode: str = "simplified",
                 guard_factory=SessionGuard, recorder_factory=Recorder,
                 toggle_key: str = "f9", stop_key: str = "f10", on_state=None):
        self._hook = hook
        self._recorder = recorder_factory(origin, mode=mode)
        self.stop_event = threading.Event()
        self._done = threading.Event()
        self._recording = False
        self._aborted_notified = False
        self._on_state = on_state
        self._guard = guard_factory(hook, self.stop_event, on_toggle=self.toggle,
                                    toggle_key=toggle_key, stop_key=stop_key)
        self._lock = threading.Lock()

    def start(self) -> None:
        """啟動監聽（同一 hook 只能 start 一次——控制與轉送共用同一條事件流）。
        缺輸入監控權限時 hook.start 可能拋例外，由面板顯示繁中提示。"""
        self._hook.start(self._dispatch)

    def toggle(self) -> None:
        """F9 或面板按鈕：開始/停止錄製。"""
        if self._done.is_set() or self.stop_event.is_set():
            return
        if self._recording:
            self._done.set()
            self._notify("done")
        else:
            self._recording = True
            self._notify("recording")

    def finish(self):
        """停止監聽並取回動作序列；放棄（stop_event）時回傳 None。"""
        self._hook.stop()
        if self.stop_event.is_set():
            return None
        return self._recorder.finish()

    def _dispatch(self, event) -> None:
        with self._lock:   # keyboard/mouse listener 各一執行緒，序列化進 Recorder
            self._guard.handle(event)   # 控制語義優先（F9 切換、F10/角落停止）
            if self.stop_event.is_set():
                if not self._aborted_notified:
                    self._aborted_notified = True
                    self._notify("aborted")
                return
            if (self._recording and not self._done.is_set()
                    and not self._guard.is_control_event(event)):
                self._recorder.feed(event)   # 自我過濾：控制鍵不錄進腳本

    def _notify(self, state: str) -> None:
        if self._on_state is not None:
            self._on_state(state)


def script_from_recording(name: str, win, actions) -> Script:
    """鏡射 cli.run_record 的目標視窗策略：優先 re.escape(標題) 精確比對；
    標題與應用程式名稱皆空時拋 ScriptValidationError（繁中）。"""
    title = re.escape(win.title) if win.title.strip() else None
    app = win.app_name if win.app_name.strip() else None
    if title is None and app is None:
        raise ScriptValidationError("目標視窗沒有標題與應用程式名稱，無法建立腳本")
    target = TargetWindow(title_pattern=title, app_name=None if title else app)
    return Script(name=name, target_window=target, actions=tuple(actions),
                  settings=ScriptSettings())
```

```python
        if state == "recording":
            self.toggle_button.setText(f"停止錄製（{self._toggle_key.upper()}）")
            self.status_label.setText(
                f"錄製中……再按 {self._toggle_key.upper()} 停止，"
                f"{self._stop_key.upper()} 放棄")
            return
        actions = self._session.finish() if self._session is not None else None
        self._session = None
        self.toggle_button.setText(f"開始錄製（{self._toggle_key.upper()}）")
        if state == "aborted" or actions is None:
            self.status_label.setText("已放棄錄製")
        elif not actions:
            self.status_label.setText("沒有錄到任何動作")
        else:
            self._actions = actions
            self.status_label.setText(f"錄到 {len(actions)} 個動作，輸入名稱後儲存")

    def _on_save_clicked(self) -> None:
        if not self._actions:
            self.status_label.setText("沒有可儲存的錄製結果")
            return
        try:
            script = script_from_recording(self.name_edit.text().strip(),
                                           self._win, self._actions)
            path = self._storage.save(script)
        except ScriptValidationError as e:
            self.status_label.setText(f"儲存失敗：{e}")   # 名稱不合法→留在面板重試
            return
        self.status_label.setText(f"已儲存：{path}（共 {len(self._actions)} 個動作）")
```

```python
_DEFAULT_LANGUAGES = ["zh-Hant", "zh-Hans", "en-US"]


class VisionOcrEngine(OcrEngine):
    def __init__(self, recognizer=None, to_image=None, languages=None):
        self._recognizer = recognizer
        self._to_image = to_image
        self._languages = list(languages) if languages else list(_DEFAULT_LANGUAGES)

    @property
    def languages(self) -> list[str]:
        return list(self._languages)

    def recognize(self, image) -> list[TextBox]:
        to_image = self._to_image or _default_to_image
        recognizer = self._recognizer or (
            lambda cg: _default_recognizer(cg, self._languages))
        cg = to_image(image)
        lines = recognizer(cg)
        return [TextBox(text=line.text, x=line.bbox[0], y=line.bbox[1],
                        width=line.bbox[2], height=line.bbox[3],
                        confidence=line.conf) for line in lines]
```
